## Track usage

Record a billable event with a stable transaction ID, and spend credits for one unit of work in a single call.

<CodeGroup>

```ts track-usage.ts
import { Nozle } from '@nozle-js/node';

const nozle = new Nozle({ apiKey: process.env.NOZLE_SECRET_KEY });

interface Generation {
  id: string;
  customerId: string;
  inputTokens: number;
  outputTokens: number;
  completedAt: Date;
}

// Bill a feature metered from events. Call it after the work succeeded.
export async function trackGeneration(generation: Generation): Promise<string> {
  // The same transactionId and timestamp on a retry record the event once.
  return nozle.track(
    generation.customerId,
    'text_generation',
    {
      input_tokens: generation.inputTokens,
      output_tokens: generation.outputTokens,
    },
    {
      transactionId: `generation:${generation.id}:completed`,
      timestamp: generation.completedAt.toISOString(),
    },
  );
}

// Spend credits for one unit of work: checked and deducted in one call.
export async function chargeAgentStep(customerId: string, stepId: string, startedAt: Date) {
  const result = await nozle.usage.track(
    {
      customerId,
      featureCode: 'agent_execution',
      creditSystemCode: 'ai_credits',
      properties: { model: 'example-model' },
      timestamp: startedAt.toISOString(),
    },
    { idempotencyKey: `agent-step:${stepId}` },
  );
  if (!result.allowed) {
    return { charged: false as const, reason: result.reason, available: result.available };
  }
  return { charged: true as const, consumed: result.credits_consumed, remaining: result.remaining };
}
```

```python track_usage.py
import os
from datetime import datetime
from typing import Any

from nozle import Nozle

nozle = Nozle(api_key=os.environ["NOZLE_SECRET_KEY"])


def track_generation(
    customer_id: str,
    generation_id: str,
    input_tokens: int,
    output_tokens: int,
    completed_at: datetime,  # timezone-aware: the API rejects a timestamp without an offset
) -> str:
    """Bill a feature metered from events. Call it after the work succeeded."""
    # The same transaction_id and timestamp on a retry record the event once.
    return nozle.track(
        customer_id,
        "text_generation",
        metadata={"input_tokens": input_tokens, "output_tokens": output_tokens},
        transaction_id=f"generation:{generation_id}:completed",
        timestamp=completed_at.isoformat(),
    )


def charge_agent_step(customer_id: str, step_id: str, started_at: datetime) -> dict[str, Any]:
    """Spend credits for one unit of work: checked and deducted in one call.

    started_at must be timezone-aware: the API rejects a timestamp without an offset.
    """
    result = nozle.usage.track(
        customer_id,
        "agent_execution",
        credit_system_code="ai_credits",
        properties={"model": "example-model"},
        timestamp=started_at.isoformat(),
        idempotency_key=f"agent-step:{step_id}",
    )
    if not result["allowed"]:
        return {"charged": False, "reason": result.get("reason"), "available": result.get("available")}
    return {"charged": True, "consumed": result.get("credits_consumed"), "remaining": result.get("remaining")}
```

## Check an entitlement

Gate a request with `can()`: deny only when `allowed` is false, and treat `overage` as billed: it never denies by itself.

<CodeGroup>

```ts check-entitlement.ts
import { Nozle } from '@nozle-js/node';

const nozle = new Nozle({ apiKey: process.env.NOZLE_SECRET_KEY });

// Gate a request on the customer's plan. customerId comes from your session, never from the browser.
export async function handleAnalyticsRequest(customerId: string): Promise<Response> {
  const check = await nozle.can(customerId, 'analytics');

  if (!check.allowed) {
    return Response.json(
      {
        error: check.reason ?? 'upgrade_required',
        used: check.used,
        limit: check.limit,
        remaining: check.remaining,
      },
      { status: 403 },
    );
  }

  // overage never denies by itself: a soft limit is reached and the usage is billed as overage.
  if (check.overage) console.warn(`${customerId} is past the analytics limit and billed as overage`);

  return Response.json(await runAnalyticsQuery(customerId));
}

// A customer with several subscriptions names the one to check.
export async function canUseSeat(customerId: string, subscriptionId: string) {
  const check = await nozle.can(customerId, 'seats', {}, { subscriptionId });
  return check.allowed;
}

async function runAnalyticsQuery(customerId: string) {
  return { customerId, rows: [] };
}
```

## Create a checkout and handle the result

Start a checkout from your backend and handle every result type: `hosted`, `razorpay`, `processing`, `completed` and `scheduled`.

<CodeGroup>

```ts checkout.ts
import { Nozle, NozleApiError, type CheckoutResult } from '@nozle-js/node';

const nozle = new Nozle({ apiKey: process.env.NOZLE_SECRET_KEY });

const allowedPlans = new Set(['pro_monthly', 'pro_annual']);
const returnUrl = 'https://app.example.com/settings/billing';

// What your backend sends back to the browser.
type CheckoutReply =
  | { action: 'redirect'; url: string; planUpgradeIntentId?: string }
  | { action: 'open_razorpay'; keyId: string; orderId: string; amountCents: number; currency: string; checkoutId: string }
  | { action: 'retry_later'; checkoutId: string }
  | { action: 'done'; planCode?: string }
  | { action: 'scheduled'; planCode?: string; effectiveAt?: string }
  | { action: 'error'; code: string };

export async function startCheckout(customerId: string, planCode: string, actionKey: string): Promise<CheckoutReply> {
  if (!allowedPlans.has(planCode)) return { action: 'error', code: 'invalid_plan' };
  try {
    const result = await nozle.checkout(customerId, planCode, returnUrl, { idempotencyKey: actionKey });
    return toReply(result);
  } catch (error) {
    if (error instanceof NozleApiError && error.status === 422 && error.hasCode('checkout_not_required')) {
      // Free, trial or pay-in-arrears plans need no payment up front.
      await nozle.subscribe(customerId, planCode);
      return { action: 'done', planCode };
    }
    if (error instanceof NozleApiError) return { action: 'error', code: error.code ?? 'checkout_failed' };
    throw error;
  }
}

function toReply(result: CheckoutResult): CheckoutReply {
  switch (result.type) {
    case 'hosted':
      // Stripe, or Nozle's auto-pay page. The return to returnUrl proves nothing:
      // confirm with the subscription.started webhook or by reloading billing state.
      // A paid upgrade carries planUpgradeIntentId: poll it with checkoutSettled.
      return { action: 'redirect', url: result.url, planUpgradeIntentId: result.plan_upgrade_intent_id };
    case 'razorpay':
      return {
        action: 'open_razorpay',
        keyId: result.key_id,
        orderId: result.order_id,
        amountCents: result.amount_cents,
        currency: result.currency,
        checkoutId: result.checkout_id,
      };
    case 'processing':
      return { action: 'retry_later', checkoutId: result.checkout_id };
    case 'completed':
      return { action: 'done', planCode: result.plan_code };
    case 'scheduled':
      return { action: 'scheduled', planCode: result.plan_code, effectiveAt: result.effective_at };
    default: {
      const unhandled: never = result;
      throw new Error(`Unhandled checkout result: ${JSON.stringify(unhandled)}`);
    }
  }
}

// Poll a `processing` result's checkout_id, or a paid upgrade's plan_upgrade_intent_id.
// A Stripe upgrade needs the customer and subscription scope.
export async function checkoutSettled(
  checkoutId: string,
  scope?: { customerId: string; subscriptionId: string },
): Promise<boolean> {
  const status = await nozle.checkoutStatus(checkoutId, scope);
  return status.status === 'succeeded' && status.fulfillment_status === 'succeeded';
}
```

```python checkout.py
from __future__ import annotations

import os
from typing import Any

from nozle import CheckoutResult, Nozle, NozleAPIError

nozle = Nozle(api_key=os.environ["NOZLE_SECRET_KEY"])

ALLOWED_PLANS = {"pro_monthly", "pro_annual"}
RETURN_URL = "https://app.example.com/settings/billing"


def start_checkout(customer_id: str, plan_code: str, action_key: str) -> dict[str, Any]:
    """Start a checkout and tell the browser what to do next."""
    if plan_code not in ALLOWED_PLANS:
        return {"action": "error", "code": "invalid_plan"}
    try:
        result = nozle.checkout(customer_id, plan_code, RETURN_URL, idempotency_key=action_key)
    except NozleAPIError as error:
        if error.status_code == 422 and error.has_code("checkout_not_required"):
            # Free, trial or pay-in-arrears plans need no payment up front.
            nozle.subscribe(customer_id, plan_code)
            return {"action": "done", "plan_code": plan_code}
        return {"action": "error", "code": error.code or "checkout_failed"}
    return to_reply(result)


def to_reply(result: CheckoutResult) -> dict[str, Any]:
    if result["type"] == "hosted":
        # Stripe, or Nozle's auto-pay page. The return to RETURN_URL proves nothing:
        # confirm with the subscription.started webhook or by reloading billing state.
        # A paid upgrade carries plan_upgrade_intent_id: poll it with checkout_settled.
        return {
            "action": "redirect",
            "url": result["url"],
            "plan_upgrade_intent_id": result.get("plan_upgrade_intent_id"),
        }
    if result["type"] == "razorpay":
        return {
            "action": "open_razorpay",
            "key_id": result["key_id"],
            "order_id": result["order_id"],
            "amount_cents": result["amount_cents"],
            "currency": result["currency"],
            "checkout_id": result["checkout_id"],
        }
    if result["type"] == "processing":
        return {"action": "retry_later", "checkout_id": result["checkout_id"]}
    if result["type"] == "completed":
        return {"action": "done", "plan_code": result.get("plan_code")}
    if result["type"] == "scheduled":
        return {
            "action": "scheduled",
            "plan_code": result.get("plan_code"),
            "effective_at": result.get("effective_at"),
        }
    raise ValueError(f"unhandled checkout result: {result!r}")


def checkout_settled(
    checkout_id: str, customer_id: str | None = None, subscription_id: str | None = None
) -> bool:
    """Poll a processing result's checkout_id, or a paid upgrade's plan_upgrade_intent_id.

    A Stripe upgrade needs the customer and subscription scope.
    """
    status = nozle.checkout_status(
        checkout_id, customer_id=customer_id, subscription_id=subscription_id
    )
    return status["status"] == "succeeded" and status["fulfillment_status"] == "succeeded"
```

## Verify and handle a webhook

Verify the Standard Webhooks signature on the raw body, skip duplicate deliveries, and act on subscription and invoice events.

<CodeGroup>

```ts webhooks.ts
import { Nozle, WebhookVerificationError, createWebhookHandler, type NozleWebhookEvent } from '@nozle-js/node';

const nozle = new Nozle({ apiKey: process.env.NOZLE_SECRET_KEY });

// The endpoint's whsec_ secret. Pass [newSecret, oldSecret] while you rotate it.
const webhookSecret = process.env.NOZLE_WEBHOOK_SECRET;

const handled = new Set<string>(); // Use your database: Nozle retries a delivery until it gets a 2xx.

async function onEvent(event: NozleWebhookEvent): Promise<void> {
  if (handled.has(event.id)) return;
  switch (event.type) {
    case 'subscription.started':
      await grantAccess(event.object);
      break;
    case 'subscription.terminated':
      await revokeAccess(event.object);
      break;
    case 'invoice.payment_status_updated':
      await recordPayment(event.object);
      break;
    default:
      break; // Unsubscribed or new event types: acknowledge and ignore.
  }
  handled.add(event.id);
}

// Fetch-style frameworks (Next.js route handlers, Remix, Hono, Bun, Deno).
// Answers 200, 400 for a forged or stale request, 405 for other methods, 500 when onEvent throws.
export const POST = createWebhookHandler(nozle, { secret: webhookSecret, onEvent });

// Any other framework: verify the raw body before parsing it.
export async function handleRawWebhook(rawBody: Buffer, headers: Record<string, string | string[] | undefined>) {
  try {
    const event = await nozle.webhooks.verify({ body: rawBody, headers, secret: webhookSecret });
    await onEvent(event);
    return { status: 200 };
  } catch (error) {
    if (error instanceof WebhookVerificationError) return { status: 400, code: error.code };
    throw error; // Answer 500 so Nozle delivers it again.
  }
}

async function grantAccess(subscription: Record<string, unknown>) {
  console.log('grant', subscription.external_id);
}
async function revokeAccess(subscription: Record<string, unknown>) {
  console.log('revoke', subscription.external_id);
}
async function recordPayment(invoice: Record<string, unknown>) {
  console.log('invoice', invoice.nozle_id, invoice.payment_status);
}
```

```python webhooks.py
from __future__ import annotations

import os
from collections.abc import Mapping
from typing import Any

from nozle import Nozle, NozleWebhookEvent, WebhookVerificationError

nozle = Nozle(api_key=os.environ["NOZLE_SECRET_KEY"])

# The endpoint's whsec_ secret. Pass [new_secret, old_secret] while you rotate it.
WEBHOOK_SECRET = os.environ["NOZLE_WEBHOOK_SECRET"]

handled: set[str] = set()  # Use your database: Nozle retries a delivery until it gets a 2xx.


def handle_webhook(raw_body: bytes, headers: Mapping[str, Any]) -> tuple[dict[str, Any], int]:
    """Verify the raw body before parsing it. Flask: request.get_data(), request.headers."""
    try:
        event = nozle.webhooks.verify(raw_body, headers, webhook_secret=WEBHOOK_SECRET)
    except WebhookVerificationError as error:
        return {"code": error.code}, 400
    on_event(event)  # An exception here answers 500, so Nozle delivers it again.
    return {"received": True}, 200


def on_event(event: NozleWebhookEvent) -> None:
    if event.id in handled:
        return
    if event.type == "subscription.started":
        grant_access(event.object)
    elif event.type == "subscription.terminated":
        revoke_access(event.object)
    elif event.type == "invoice.payment_status_updated":
        record_payment(event.object)
    # Other event types: acknowledge and ignore.
    handled.add(event.id)


def grant_access(subscription: dict[str, Any]) -> None:
    print("grant", subscription.get("external_id"))


def revoke_access(subscription: dict[str, Any]) -> None:
    print("revoke", subscription.get("external_id"))


def record_payment(invoice: dict[str, Any]) -> None:
    print("invoice", invoice.get("nozle_id"), invoice.get("payment_status"))
```

## Change or cancel a subscription

Preview a plan change, apply the confirmed quote (an upgrade through checkout, a downgrade as a transition), cancel at period end, keep, or cancel now.

<CodeGroup>

```ts change-subscription.ts
import { Nozle, type SubscriptionChangePreview } from '@nozle-js/node';

const nozle = new Nozle({ apiKey: process.env.NOZLE_SECRET_KEY });

const returnUrl = 'https://app.example.com/settings/billing';

// Step 1: the plans this subscription may move to, and the exact price of one move.
export async function previewPlanChange(customerId: string, subscriptionId: string, targetPlanCode: string) {
  const options = await nozle.subscriptionOptions(customerId, subscriptionId);
  if (!options.eligible_plans.some((plan) => plan.code === targetPlanCode)) {
    throw new Error(`${targetPlanCode} is not an eligible plan for ${subscriptionId}`);
  }
  // Show these amounts and dates to the customer before they confirm. The quote lasts five minutes.
  return nozle.previewSubscriptionChange(customerId, subscriptionId, targetPlanCode);
}

// Step 2: apply the confirmed quote. actionKey is stable per confirmation, so a retry replays it.
export async function confirmPlanChange(
  customerId: string,
  subscriptionId: string,
  targetPlanCode: string,
  quote: SubscriptionChangePreview,
  actionKey: string,
) {
  if (quote.transition_direction === 'upgrade') {
    // A paid upgrade goes through checkout; handle the result as in checkout.ts.
    return nozle.checkout(customerId, targetPlanCode, returnUrl, {
      subscriptionId,
      quoteId: quote.quote_id,
      idempotencyKey: actionKey,
    });
  }
  return nozle.applySubscriptionTransition(
    {
      customerId,
      subscriptionId,
      operation: 'downgrade',
      targetPlanCode,
      timing: quote.timing,
      billingAnchor: 'keep_anchor',
      quoteId: quote.quote_id,
    },
    actionKey,
  );
}

// Cancel at the end of the period, after the customer saw the ending date.
export async function cancelAtPeriodEnd(customerId: string, subscriptionId: string, actionKey: string) {
  const preview = await nozle.previewSubscriptionTransition({
    customerId,
    subscriptionId,
    operation: 'cancel',
    timing: 'end_of_period',
  });
  return nozle.applySubscriptionTransition(
    {
      customerId,
      subscriptionId,
      operation: 'cancel',
      timing: 'end_of_period',
      expectedEffectiveAt: preview.subscription_transition.effective_at,
    },
    actionKey,
  );
}

// Undo a scheduled cancellation before it takes effect.
export async function keepSubscription(customerId: string, subscriptionId: string, actionKey: string) {
  return nozle.applySubscriptionTransition({ customerId, subscriptionId, operation: 'uncancel' }, actionKey);
}

// End it now, for example when an admin closes the account.
export async function cancelNow(customerId: string, subscriptionId: string) {
  const { subscription } = await nozle.cancelSubscription(customerId, subscriptionId, { policy: 'immediate' });
  return subscription.status;
}
```

```python change_subscription.py
    """Step 2: apply the confirmed quote. action_key is stable per confirmation, so a retry replays it."""
    if quote["transition_direction"] == "upgrade":
        # A paid upgrade goes through checkout; handle the result as in checkout.py.
        return nozle.checkout(
            customer_id,
            target_plan_code,
            RETURN_URL,
            subscription_id=subscription_id,
            quote_id=quote["quote_id"],
            idempotency_key=action_key,
        )
    return nozle.apply_subscription_transition(
        {
            "customer_id": customer_id,
            "subscription_id": subscription_id,
            "operation": "downgrade",
            "target_plan_code": target_plan_code,
            "timing": quote["timing"],
            "billing_anchor": "keep_anchor",
            "quote_id": quote["quote_id"],
        },
        idempotency_key=action_key,
    )


def cancel_at_period_end(customer_id: str, subscription_id: str, action_key: str) -> Any:
    """Cancel at the end of the period, after the customer saw the ending date."""
    preview = nozle.preview_subscription_transition(
        {
            "customer_id": customer_id,
            "subscription_id": subscription_id,
            "operation": "cancel",
            "timing": "end_of_period",
        }
    )
    return nozle.apply_subscription_transition(
        {
            "customer_id": customer_id,
            "subscription_id": subscription_id,
            "operation": "cancel",
            "timing": "end_of_period",
            "expected_effective_at": preview["subscription_transition"]["effective_at"],
        },
        idempotency_key=action_key,
    )


def keep_subscription(customer_id: str, subscription_id: str, action_key: str) -> Any:
    """Undo a scheduled cancellation before it takes effect."""
    return nozle.apply_subscription_transition(
        {"customer_id": customer_id, "subscription_id": subscription_id, "operation": "uncancel"},
        idempotency_key=action_key,
    )


def cancel_now(customer_id: str, subscription_id: str) -> str:
    """End it now, for example when an admin closes the account."""
    return nozle.cancel_subscription(customer_id, subscription_id, "immediate")["subscription"]["status"]
```
